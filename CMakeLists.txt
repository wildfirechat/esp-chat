cmake_minimum_required(VERSION 3.16)

# sdkconfig.local holds WiFi credentials and the WFC test account -- gitignored,
# see README. Later files in SDKCONFIG_DEFAULTS win, so it overrides
# sdkconfig.defaults. The EXISTS guard matters: IDF treats a file listed in
# SDKCONFIG_DEFAULTS but missing on disk as a fatal error.
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults")
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.local")
    list(APPEND SDKCONFIG_DEFAULTS "sdkconfig.local")
endif()

# This is the application. The three things it is an application of are
# separate modules, and each of them arrives here in one of two forms:
#
#   ../wfc-esp/components    source, when the sibling repositories are checked
#   ../wfav-esp/components   out beside this project. That is how the modules
#   ../wfptt-esp/components  themselves are developed -- one edit, one rebuild --
#                            and each of them can still be read, built and
#                            tested on its own.
#
#   sdk/wfc, sdk/wfav,       the same three modules packaged as binary
#   sdk/wfptt                components: libwfc.a plus the public headers, and
#                            no source at all. Generated by tools/pack_sdk.py;
#                            see sdk/README.md.
#
# The package wins wherever both are present, and that way round matters: the
# published tree is a sibling of the modules in the maintainer's own workspace,
# so "use the source if it happens to be next door" would mean the .a that is
# about to be published is the one thing never built or tested. sdk/ is a
# deliberate artifact of the tree it sits in; a checkout beside it is ambient.
# Which form is in use is the ONLY structural difference between the internal
# tree and the published one -- this file, main/, components/ and the partition
# table are the same bytes in both, which is what makes a merge between them a
# copy. Both branches say which one they took, because getting this wrong is
# silent otherwise.
#
#   ../wfc-esp    the IM client -- protocol, store, model, events. Required.
#   ../wfav-esp   the AV SDK, which depends on wfc-esp the way avenginekit.js
#                 depends on WFC.js. Optional: without it the board is a chat
#                 client with no 通话 button.
#   ../wfptt-esp  the push-to-talk SDK, which depends on wfc-esp the way ptt.js
#                 depends on WFC.js. Optional and INDEPENDENT of wfav: a talk
#                 is AMR-NB over the long link and wants nothing from WebRTC,
#                 so the three combinations that leave one of them out are all
#                 real builds.
#
# Both of the optional two are optional in two independent ways, and both have
# to be handled here because EXTRA_COMPONENT_DIRS is decided before sdkconfig
# exists:
#
#   - it may not be provided in either form (someone cloned wfc-esp and this,
#     and nothing else), which is why those branches below have no else;
#   - it may be provided and still not wanted, which is CONFIG_APP_CALL=n /
#     CONFIG_APP_PTT=n. That one is handled a level down, in
#     main/CMakeLists.txt.
#
# Leaving one on the component path when its switch is n costs the download of
# its managed components but nothing in the image: nothing REQUIREs it, so it
# is never built or linked. Deleting sdk/<name> -- or not cloning the sibling
# -- is what avoids the download too. For wfav that download is esp_peer,
# esp_capture and the Opus codecs; for wfptt there is nothing to download at
# all, which is one measure of how much smaller it is.
set(wfc_source "${CMAKE_CURRENT_LIST_DIR}/../wfc-esp/components")
set(wfav_source "${CMAKE_CURRENT_LIST_DIR}/../wfav-esp/components")
set(wfptt_source "${CMAKE_CURRENT_LIST_DIR}/../wfptt-esp/components")

if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdk/wfc")
    # The component itself, not a directory of components: pbc and sqlite3 are
    # already inside libwfc.a, so there is nothing beside it to pick up.
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/sdk/wfc")
    message(STATUS "wfc: sdk/wfc (packaged .a -- see sdk/README.md)")
elseif(EXISTS "${wfc_source}/wfc")
    # The directory rather than the component: wfc comes with pbc and sqlite3.
    list(APPEND EXTRA_COMPONENT_DIRS "${wfc_source}")
    message(STATUS "wfc: ../wfc-esp/components (source)")
else()
    message(FATAL_ERROR
        "No IM client to build against: there is neither sdk/wfc (package) nor "
        "../wfc-esp/components/wfc (source). Clone wfc-esp beside this project, "
        "or run tools/pack_sdk.py to generate sdk/.")
endif()

if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdk/wfav")
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/sdk/wfav")
    message(STATUS "wfav: sdk/wfav (packaged .a)")
elseif(EXISTS "${wfav_source}/wfav")
    list(APPEND EXTRA_COMPONENT_DIRS "${wfav_source}")
    message(STATUS "wfav: ../wfav-esp/components (source)")
else()
    message(STATUS "wfav: not provided -- CONFIG_APP_CALL must be n")
endif()

if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdk/wfptt")
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/sdk/wfptt")
    message(STATUS "wfptt: sdk/wfptt (packaged .a)")
elseif(EXISTS "${wfptt_source}/wfptt")
    list(APPEND EXTRA_COMPONENT_DIRS "${wfptt_source}")
    message(STATUS "wfptt: ../wfptt-esp/components (source)")
else()
    message(STATUS "wfptt: not provided -- CONFIG_APP_PTT must be n")
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

# __FILE__ out of the modules, which are outside both directories IDF's
# CONFIG_COMPILER_HIDE_PATHS_MACROS knows how to rewrite ($IDF_PATH and this
# project), would otherwise reach the image as an absolute path: one assert()
# in pbc is enough to put a developer's home directory into a published .a.
# Map the directory the modules sit in to nothing, so those come out relative.
get_filename_component(workspace_dir "${CMAKE_CURRENT_LIST_DIR}" DIRECTORY)
idf_build_set_property(COMPILE_OPTIONS "-fmacro-prefix-map=${workspace_dir}/=" APPEND)

project(esp_chat)
